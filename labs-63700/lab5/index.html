def load_image(path):
    """Read an 8-bit grayscale image as an (H, W) uint8 array."""
    return np.ascontiguousarray(np.asarray(Image.open(path).convert("L"), dtype=np.uint8))


def save_image(x, path):
    """Write an (H, W) uint8 array as an 8-bit grayscale PNG."""
    Image.fromarray(np.asarray(x, dtype=np.uint8)).save(path)


def save_label_image(labels, path):
    """Write a label image as a color PNG: label 0 black, others fixed random colors."""
    k = int(labels.max())
    colors = np.random.default_rng(0).integers(40, 256, size=(k + 1, 3), dtype=np.uint8)
    colors[0] = 0
    Image.fromarray(colors[labels]).save(path)

Step 2: Neighborhoods and Connected Sets

Let \(x_s\) be the pixel value at the lattice point \(s = (m, n)\). We use a 4 point neighborhood, so the neighbors of \(s\) are

$$ \partial s = \big\{ (m-1, n), \ (m+1, n), \ (m, n-1), \ (m, n+1) \big\} \ . \tag{1} $$

We also use a free boundary, so pixels along the boundary of the image have fewer than 4 neighbors each. Two neighboring lattice points \(r \in \partial s\) are connected neighbors if \(|x_s - x_r| \le T\), where \(T\) is a fixed threshold. We denote the connected neighbors of \(s\) by the set

$$ c(s) = \big\{ r \in \partial s \ : \ |x_s - x_r| \le T \big\} \ . \tag{2} $$

Pixels \(s\) and \(r\) are connected if there is a sequence of pixels from \(s\) to \(r\) in which each pixel is a connected neighbor of the one before it. The connected set of a seed \(s_0\) is the set of all pixels connected to it. A connected component is a set that is the connected set of each of its pixels.

The connected set may be found by maintaining 1) a list of pixels \(B\) which are known to be connected to \(s_0\) but have not yet been searched, and 2) a segmentation image \(Y\) which marks the pixels known to be connected to \(s_0\).

Region growing from the seed \(s_0\).
  1. Set \(Y_{s_0} \leftarrow\) label and \(B \leftarrow \{s_0\}\).
  2. While \(B\) is not empty:
    1. Remove any pixel \(s\) from \(B\).
    2. For each \(r \in c(s)\) whose \(Y_r\) is not yet the label, set \(Y_r \leftarrow\) label and add \(r\) to \(B\).

This type of algorithm is referred to as region growing, because the segmented region grows out from the seed point \(s_0\). At each step, \(B\) contains pixels at the boundary of the set whose neighbors have not yet been checked. A pixel is labeled when it enters \(B\), so it enters \(B\) at most once.

Step 7: Call the C Function from Python

Copy these functions into lab5.py. They use ctypes to call the library. connected_set_c has the same signature as connected_set, and segment_c returns the same values as segment. 

_LIB = None


def load_connect_library(path=None):
    """Load libconnect.so (default: next to this file) once and declare its argument types."""
    global _LIB
    if _LIB is None:
        if path is None:
            path = os.path.join(os.path.dirname(os.path.abspath(__file__)), "libconnect.so")
        lib = ctypes.CDLL(path)
        lib.connected_set.restype = ctypes.c_int
        lib.connected_set.argtypes = [
            ctypes.c_void_p, ctypes.c_int, ctypes.c_int, ctypes.c_int,
            ctypes.c_int, ctypes.c_int, ctypes.c_int, ctypes.c_void_p]
        lib.segment_image.restype = ctypes.c_int
        lib.segment_image.argtypes = [
            ctypes.c_void_p, ctypes.c_int, ctypes.c_int, ctypes.c_int,
            ctypes.c_int, ctypes.c_void_p, ctypes.POINTER(ctypes.c_int)]
        _LIB = lib
    return _LIB


def connected_set_c(x, seed, T, label, seg):
    """Same as connected_set, using C; x is C-contiguous uint8, seg C-contiguous int32."""
    if x.dtype != np.uint8 or seg.dtype != np.int32 \
            or not x.flags.c_contiguous or not seg.flags.c_contiguous \
            or x.shape != seg.shape:
        raise TypeError("x must be C-contiguous uint8 and seg C-contiguous int32 of the same shape")
    lib = load_connect_library()
    H, W = x.shape
    count = lib.connected_set(x.ctypes.data, H, W, int(seed[0]), int(seed[1]),
                              int(T), int(label), seg.ctypes.data)
    if count < 0:
        raise MemoryError("connected_set in C could not allocate its list")
    return count


def segment_c(x, T, min_size):
    """Same as segment(x, T, min_size, connected_set_fn), in one call to C."""
    if x.dtype != np.uint8 or not x.flags.c_contiguous:
        raise TypeError("x must be a C-contiguous uint8 array")
    lib = load_connect_library()
    H, W = x.shape
    labels = np.empty((H, W), dtype=np.int32)
    n_all = ctypes.c_int(0)
    K = lib.segment_image(x.ctypes.data, H, W, int(T), int(min_size),
                          labels.ctypes.data, ctypes.byref(n_all))
    if K < 0:
        raise MemoryError("segment_image in C could not allocate its arrays")
    return labels, K, n_all.value

C receives only the address x.ctypes.data, not the shape or type. The wrappers check them, because a wrong type or non-contiguous array makes C read the wrong bytes with no error.
